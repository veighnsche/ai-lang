func TestAuditCanonicalEvidenceProbe(t *testing.T) {
	src := `mod acc
  provides [acc__a, acc__b, acc__factory]
  uses []
  emits []
fn acc__a(value: int) -> int rev 1
  emits []
  tests
    one(1) => Ok(1)
  Ok(value)
fn acc__b(value: int) -> int rev 1
  emits []
  tests
    one(1) => Ok(2)
  Ok(value + 1)
fn acc__factory() -> Fn<int, int, []> rev 1
  emits []
  tests
    one() => Ok(fnref acc__a()) pinned
  Ok(fnref acc__a())
`
	p, _, ds := genericProgram(t, map[string]string{"acc.can": src})
	if len(genericErrs(ds)) > 0 {
		t.Fatal(ds)
	}
	base := acceptanceBase(t, p)
	moved := strings.ReplaceAll(src, "fnref acc__a()", "fnref acc__b()")
	q, _, ds := genericProgram(t, map[string]string{"acc.can": moved})
	if len(genericErrs(ds)) > 0 {
		t.Fatal(ds)
	}
	old := canonPinnedRow(p.Fns["acc__factory"].Tests[0])
	new := canonPinnedRow(q.Fns["acc__factory"].Tests[0])
	warnings := CheckPinnedRows(q, map[string]string{"acc.can": moved}, base)
	t.Logf("PIN_TARGET_COLLISION old=%s new=%s warnings=%v", old, new, warnings)
	if old != new || len(warnings) != 0 {
		t.Fatal("hypothesis falsified: inspect result")
	}
	a, _ := parseSmall("Ok<int>(value)")
	b, _ := parseSmall("Ok<str>(value)")
	t.Logf("TYPED_OK_CANON_EQUAL=%v", canonSmall(a) == canonSmall(b))
	pa, _ := parsePattern("Ok<int> r")
	pb, _ := parsePattern("Ok<str> r")
	t.Logf("TYPED_PATTERN_CANON_EQUAL=%v", canonPattern(pa) == canonPattern(pb))
	n := &Node{IsMatch: true, Kind: MatchInvoke, Scruts: []*Small{{Kind: "ref", Ref: []string{"cb"}}}}
	n.InvokeArg, _ = parseSmall("1")
	first := canonNode(n)
	n.InvokeArg, _ = parseSmall("2")
	t.Logf("INVOKE_ARGUMENT_CANON_EQUAL=%v", first == canonNode(n))
}

func TestAuditDuplicateRecordProbe(t *testing.T) {
	files := map[string]string{
		"a.can": "mod a\n  provides [Shared__Record]\n  uses []\n  emits []\ntype Shared__Record rev 1 (\n  value: int\n)\n",
		"b.can": "mod b\n  provides [Shared__Record]\n  uses []\n  emits []\ntype Shared__Record rev 1 (\n  value: str\n)\n",
	}
	p, _, ds := genericProgram(t, files)
	t.Logf("DUPLICATE_RECORD diagnostics=%v shape=%v", genericErrs(ds), recordShapes(p.Modules)["Shared__Record"])
	if len(genericErrs(ds)) != 0 {
		t.Fatal("hypothesis falsified: inspect diagnostics")
	}
	files["a.can"], files["b.can"] = files["b.can"], files["a.can"]
	p, _, ds = genericProgram(t, files)
	t.Logf("DUPLICATE_RECORD_REORDERED diagnostics=%v shape=%v", genericErrs(ds), recordShapes(p.Modules)["Shared__Record"])
}
